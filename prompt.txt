# 🔚 INPUT SCRIPT

import time
import re
import csv
import os
import json
import pandas as pd
from tqdm import tqdm

from selenium import webdriver
from selenium.webdriver.common.by import By
from selenium.webdriver.support.ui import Select, WebDriverWait
from selenium.webdriver.support import expected_conditions as EC
from selenium.common.exceptions import StaleElementReferenceException

from webdriver_manager.chrome import ChromeDriverManager
from selenium.webdriver.chrome.service import Service


OUTPUT_FILE = "jkpcb_full_data.csv"
CHECKPOINT_FILE = "jkpcb_checkpoint.json"
LOG_FILE = "jkpcb_log.txt"

YEARS = ["2022", "2023", "2024", "2025", "2026"]


class JKPCBScraper:

    def __init__(self):
        options = webdriver.ChromeOptions()
        options.add_argument("--headless=new")
        options.add_argument("--window-size=1920,1080")

        self.driver = webdriver.Chrome(
            service=Service(ChromeDriverManager().install()),
            options=options
        )

        self.wait = WebDriverWait(self.driver, 25)
        self.url = "https://jkpcb.jk.gov.in/airquality.aspx"

        self.pattern = re.compile(r"([A-Za-z]+-\d{4})\s+(\d+)")

        self.parameters = {
            "aqi": "AQI",
            "pm10": "PM10",
            "pm25": "PM2_5",
            "no2": "NO2",
            "so2": "SO2"
        }

        self.months = [
            "January","February","March","April","May","June",
            "July","August","September","October","November","December"
        ]

        self._init_files()

    # -------------------------------
    # FILE INIT
    # -------------------------------
    def _init_files(self):
        if not os.path.exists(OUTPUT_FILE):
            with open(OUTPUT_FILE, "w", newline="") as f:
                writer = csv.writer(f)
                writer.writerow([
                    "Region","District","Station","Year","Month",
                    "AQI","PM10","PM2_5","NO2","SO2"
                ])

        if not os.path.exists(CHECKPOINT_FILE):
            with open(CHECKPOINT_FILE, "w") as f:
                json.dump({}, f)

    def log(self, msg):
        with open(LOG_FILE, "a") as f:
            f.write(f"{time.ctime()} | {msg}\n")

    def save_checkpoint(self, data):
        with open(CHECKPOINT_FILE, "w") as f:
            json.dump(data, f)

    def load_checkpoint(self):
        try:
            with open(CHECKPOINT_FILE) as f:
                return json.load(f)
        except:
            return {}

    # -------------------------------
    # WAIT CHART
    # -------------------------------
    def wait_chart_loaded(self):
        self.wait.until(
            lambda d: any(
                "December-" in (el.get_attribute("aria-label") or "")
                for el in d.find_elements(By.CSS_SELECTOR, "#chartdiv12 g[aria-label]")
            )
        )
        time.sleep(1)

    # -------------------------------
    # EXTRACT
    # -------------------------------
    def extract_chart(self):
        data = {}

        elements = self.driver.find_elements(By.CSS_SELECTOR, "#chartdiv12 g[aria-label]")

        for el in elements:
            try:
                if not el.is_displayed():
                    continue

                text = el.get_attribute("aria-label")
                match = self.pattern.search(text)

                if match:
                    month_year = match.group(1)
                    value = int(match.group(2))

                    month, year = month_year.split("-")
                    key = (month, year)

                    data[key] = value

            except StaleElementReferenceException:
                continue

        return data

    # -------------------------------
    # CLICK SHOW
    # -------------------------------
    def click_show(self):
        btn = self.wait.until(EC.element_to_be_clickable((By.ID, "btnshow")))
        self.driver.execute_script("arguments[0].scrollIntoView({block:'center'});", btn)
        time.sleep(0.3)
        self.driver.execute_script("arguments[0].click();", btn)

    # -------------------------------
    # RESET CHART
    # -------------------------------
    def reset_chart(self):
        self.driver.execute_script("""
            var chart = document.getElementById('chartdiv12');
            if(chart){ chart.innerHTML = ''; }
        """)
        time.sleep(0.3)

    # -------------------------------
    # APPEND CSV
    # -------------------------------
    def append_rows(self, rows):
        with open(OUTPUT_FILE, "a", newline="") as f:
            writer = csv.writer(f)
            for r in rows:
                writer.writerow(r)

    # -------------------------------
    # MAIN RUN
    # -------------------------------
    def run(self):
        self.driver.get(self.url)

        checkpoint = self.load_checkpoint()

        region_select = Select(self.wait.until(
            EC.presence_of_element_located((By.ID, "ddregion"))
        ))

        regions = [
            opt.get_attribute("value")
            for opt in region_select.options if opt.get_attribute("value") != "Select"
        ]

        for r_i, region in enumerate(tqdm(regions, desc="Regions")):

            if checkpoint.get("region") and region < checkpoint["region"]:
                continue

            try:
                Select(self.driver.find_element(By.ID, "ddregion")).select_by_value(region)
                self.wait.until(EC.presence_of_element_located((By.ID, "dddistrict")))
                time.sleep(1)

                districts = [
                    opt.get_attribute("value")
                    for opt in Select(self.driver.find_element(By.ID, "dddistrict")).options
                    if opt.get_attribute("value") != "Select"
                ]

                for district in tqdm(districts, desc="Districts", leave=False):

                    if checkpoint.get("district") and district < checkpoint["district"]:
                        continue

                    Select(self.driver.find_element(By.ID, "dddistrict")).select_by_value(district)
                    self.wait.until(EC.presence_of_element_located((By.ID, "ddstation")))
                    time.sleep(1)

                    stations = [
                        opt.get_attribute("value")
                        for opt in Select(self.driver.find_element(By.ID, "ddstation")).options
                        if opt.get_attribute("value") != "Select"
                    ]

                    for station in tqdm(stations, desc="Stations", leave=False):

                        Select(self.driver.find_element(By.ID, "ddstation")).select_by_value(station)
                        time.sleep(1)

                        for year in YEARS:

                            month_data = {
                                m: {
                                    "AQI": "null",
                                    "PM10": "null",
                                    "PM2_5": "null",
                                    "NO2": "null",
                                    "SO2": "null"
                                } for m in self.months
                            }

                            Select(self.driver.find_element(By.ID, "ddyr1")).select_by_value(year)
                            time.sleep(1)

                            for param, col in self.parameters.items():

                                try:
                                    self.reset_chart()

                                    Select(self.driver.find_element(By.ID, "ddparam")).select_by_value(param)
                                    time.sleep(0.5)

                                    self.click_show()
                                    self.wait_chart_loaded()

                                    data = self.extract_chart()

                                    for (month, yr), value in data.items():
                                        if yr == year:
                                            if value == 0:
                                                value = "null"
                                            month_data[month][col] = value

                                except Exception as e:
                                    self.log(f"PARAM FAIL {region}-{district}-{station}-{year}-{param} | {e}")

                            # save rows immediately
                            rows = []
                            for month in self.months:
                                row = [
                                    region, district, station, year, month,
                                    month_data[month]["AQI"],
                                    month_data[month]["PM10"],
                                    month_data[month]["PM2_5"],
                                    month_data[month]["NO2"],
                                    month_data[month]["SO2"]
                                ]
                                rows.append(row)

                            self.append_rows(rows)

                            # checkpoint update
                            self.save_checkpoint({
                                "region": region,
                                "district": district,
                                "station": station,
                                "year": year
                            })

            except Exception as e:
                self.log(f"FAIL {region} | {e}")

        self.driver.quit()
        print("✅ DONE")


if __name__ == "__main__":
    JKPCBScraper().run()
